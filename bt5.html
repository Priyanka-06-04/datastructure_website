        //Binary search tree dynamically 

            #include<stdio.h>
            #include<stdlib.h>
            
            struct tree{
                int data;
                struct tree *left;
                struct tree *right;
            };
            struct tree *p,*start,*nn,*search,*prev,*root,*temp;
            
            //code to find inoreder successor
            
            struct tree* find_minimum(struct tree * root) {
              while(root -> left != NULL) 
                  root=root->left;
              return root;
            }
            
            //Code for insertion
            
            void insert(){
                int value;
                printf("Enter the value of node which you want to insert:");
                scanf("%d",&value);
                if(start==NULL){
                   start=(struct tree*)malloc(sizeof(struct tree));
                    start->data=value;
                    start->left=start->right=NULL;
                }
               else{
                   nn=(struct tree*)malloc(sizeof(struct tree));
                   nn->data=value;
                   nn->left=nn->right=NULL;
                   prev=NULL;
                   search=start;
                   
                   while(search!=NULL){
                  if(search->data>value){
                      prev=search;
                      search=search->left;
                  }
                 else if(search->data<value){
                     prev=search;
                     search=search->right;
                 }
               }
                  if(prev->data>value){
                  printf("inserted at left\n");
                  prev->left=nn;}
                  if(prev->data<value){
                  printf("inserted at right\n");
                  prev->right=nn;
               }
                   
               }
            }
            
            //Code for deletion
            
            struct tree* del(struct tree * root, int x) {
              if (root == NULL)
                return NULL;
                else{
              if (x > root -> data)
                root -> right = del(root -> right, x);
              else if (x < root -> data)
                root -> left = del(root -> left, x);
              else {
                //No Child node
                
                if (root -> left == NULL && root -> right== NULL) {
                  free(root);
                  return NULL;
                }
                else if (root -> left == NULL || root -> right == NULL) {
                  struct node *temp;
                  if (root -> left == NULL)
                    temp = root -> right;
                  else{
                    temp = root -> left;
                  free(root);
                  return temp;}
                }
              else {
                  struct tree *temp = find_minimum(root -> right);
                  root -> data = temp -> data;
                  root -> right= del(root -> right, temp -> data);
                }
              }}
              return root;
            }
            
            //       Search..
            
             void find(struct tree* root,int x){
                 if(root->data==x){
                printf("Value found");
                 return;
                 }
                else if(root->data>x)
                  find(root->left,x);
                else
                  find (root->right,x);
             }
            
            //       Inorder traversal
            
            void inorder(struct tree *root) {
              if (root != NULL) 
              {
                inorder(root -> left); 
                printf(" %d ", root -> data);
                inorder(root -> right);  
              }
            }
            
            //      Preorder traversal
            
            void preorder(struct tree *root) {
              if (root != NULL) 
              {
                printf(" %d ", root -> data);
                 preorder(root -> left); 
                preorder(root -> right);  
              }
            }
            
            //      Postorder traversal
            
            void postorder(struct tree *root) {
              if (root != NULL) 
              {
                 postorder(root -> left); 
                postorder(root -> right);  
                printf(" %d ", root -> data);
              }
            }
