Convert a number into binary using Recursion

            //decimal to binary
                int convert(int);
                #include
                int main(){
                    int dec,bin;
                    printf("Enter the decimal number:");
                    scanf("%d",&dec);
                    bin=convert(dec);
                    printf("The binary of %d is %d ",dec,bin);
                  
                }
                int convert(int dec){
                    if(dec==0)
                    return 0;
                    else
                    return (dec%2+10*convert(dec/2));
                
                }
