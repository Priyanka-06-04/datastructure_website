                char strtosent(char *str){
                    int i=0;
                    while(str[i]!='\0'){
                        if(i==0&&str[i]>='a'&&str[i]<='z')
                        str[i]=str[i]-32;
                        else if(str[i]=='.'){
                            if(str[i+1]==' ')
                            str[i+2]=str[i+2]-32;
                            else
                             str[i+1]=str[i+1]-32;
                        }
                        i++;
                    }
                }
