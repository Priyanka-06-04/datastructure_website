Enter a Integer Array From the user,perform insertion sort algorithm through iteration.

        #include<stdio.h>
            int main(){
                int arr[5],i,j;
                printf("Enter the five elements");
                for(i=0;i<5;i++){
                    scanf("%d",&arr[i]);
                }
                printf("After insertion sort");
                for(i=1;i<5;i++){
                    int current=arr[i];
                    j=i-1;
                    while(arr[j]>current&&j>=0){
                        arr[j+1]=arr[j];
                        j--;
                    }
                     arr[j+1]=current;
                }
                for(i=0;i<5;i++){
                    printf("%d ",arr[i]);
                }
            }
