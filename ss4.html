Enter a String From the user, enter the character and perform linear search algorithm through iteration.

         
            #include <stdio.h>
                int ls(char arr[], char key)
                {
                    for (int i = 0; arr[i] != '\0'; i++)
                    {
                        if (arr[i] == key)
                            return i;
                    }
                    return -1;
                }
                int main()
                {
                char str[10], key;
                int loc;
                printf("Enter the string:");
                gets(str);
                printf("Enter the element you want to search:");
                scanf("%c", &key);
                loc = ls(str, key);
            if (loc == -1)
                 printf("%c not found in the array", key);
            else
                 printf("%c found at index %d in the string",key,loc);
                return 0;
                    
                }
