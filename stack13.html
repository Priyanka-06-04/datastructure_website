Create a program to evaluate the postfix expression

        //Evaluate infix to prefix.
            #include<stdio.h>
            #include<string.h>
            void infToPos(char[],char[]);
            char stack[20];
            int index=-1;
            int isoperator(char symbol){
                switch(symbol){
                    case'+':
                    case'-':
                    case'*':
                    case'/':
                    case'(':
                    case')':        
                            return 1;
                    default:
                            return 0;    
                }
            }
            int precedence(char symbol){
                int p;
                switch(symbol){
                    case'*':
                    case'/':
                    p=3;
                    break;
                    case'+':
                    case'-':
                    p=2;
                    break;
                    case'(':
                    case')':
                    case'#':
                    p=1;
                    break;
                }
                return p;
            }
            int push(char item){
                 index++;
                 stack[index]=item;
            }
            int pop(){
                char a;
                a=stack[index];
                index--;
                return a;
            }
            void infToPos(char infix[],char postfix[]){
                int i,symbol,j=0;
                stack[++index]='#';
                for(i=0;infix[i]!='\0';i++){
                    symbol=infix[i];
                if(isoperator(symbol)==0)
                {
                    postfix[j]=symbol;
                    j++;
                }
                else{
                if(symbol=='(')
                    push(symbol);
                else if(symbol==')'){
                    while(stack[index]!='('){
                        postfix[j]=pop();
                        j++;
                    }
                    pop();
                } 
                else
                {
                    if(precedence(symbol)>precedence(stack[index]))
                    push(symbol);
                else
                 {
                    while(precedence(symbol)<=precedence(stack[index])){
                        postfix[j]=pop();
                        j++;
                    }
                    push(symbol);
                 }
                }
                }
                }
                while(stack[index]!='#'){
                    postfix[j]=pop();
                    j++;
                }
                postfix[j]='\0';
            
            }
            int evaluate(int op1,int op2,char ch){
                int n;
                printf("op1=%d\nop2=%d\noperator=%c\n",op1,op2,ch);
                switch(ch){
                    case'+':
                    n=op1+op2;
                    break;
                    case'-':
                    n=op1-op2;
                    break;
                    case'*':
                    n=op1*op2;
                    break;
                    case'/':
                    n=op1/op2;
                    break;
                    default:
                    printf("not valid opertor");
                }
                return n;
            }
            int main(){
                char infix[20],posfix[20],*revinf,*prefix,ch;
                int i,op1,op2,n;
                printf("Enter the infix expression");
                gets(infix);
                revinf=strrev(infix);
                puts(revinf);
                infToPos(revinf,posfix);
                printf("postfix:");
                puts(posfix);
                prefix=strrev(posfix);
                printf("prefix:");
                puts(prefix); 
               for(i=strlen(prefix);i>=0;i--){
                    ch = prefix[i];
                    // printf("the char is:%c",ch);
                    if(isoperator(ch)==0){
                     n=ch-'0';   
                     push(n);
                    }
                    else{
                    op1=pop();
                    op2=pop();
                    n=evaluate(op1,op2,ch);
                    push(n); 
                    }
                }
                printf("answer:%d",pop());
            }
