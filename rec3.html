Calculate the sum of natural number using Recursio

            //sum of 10 natural numbers
                #include<stdio.h>
                int sum(int);
                int main(){
                    int n=0;
                   printf("sum of 10 natural numbers: %d",sum(n));
                    
                }
                int sum(int n){
                    int add=0;
                    if(n<=10)
                    {
                        add=n+sum(n+1);
                        n++;
                    }
                    return add;
                }
