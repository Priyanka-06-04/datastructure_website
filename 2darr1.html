Enter the matrix of two rows and four columns from the user and display it in a matrix format.

        #include <stdio.h>
            void matrix(int arr[3][2], int r, int c)
            {
                int i, j;
            
                for (i = 0; i < r; i++)
                {
                    printf("Enter the elements of %d row:", i + 1);
                    for (j = 0; j < c; j++)
                    {
                        scanf("%d", &(arr[i][j]));
                    }
                }
            }
            int main()
            {
                int arr[3][2], i, j;
                int rows = 3, cols = 2;
                matrix(arr, rows, cols);
                printf("The matrix is:\n");
                for (i = 0; i < rows; i++)
                {
                    for (j = 0; j < cols; j++)
                    {
                        printf("%d\t", arr[i][j]);
                    }
                    printf("\n");
                }
                return 0;
            }

OUTPUT
