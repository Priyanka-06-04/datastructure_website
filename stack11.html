Enter the infix expression from user and convert into postfix expression

        #include<stdio.h>
            #include<nio.h>
            #include<string.h>
            void infToPos(char[],char[]);
            char stack[20];
            int index=-1;
            int isoperator(char symbol){
                switch(symbol){
                    case'+':
                    case'-':
                    case'*':
                    case'/':
                    case'(':
                    case')':        
                            return 1;
                    default:
                            return 0;    
                }
            }
            int precedence(char symbol){
                int p;
                switch(symbol){
                    case'*':
                    case'/':
                    p=3;
                    break;
                    case'+':
                    case'-':
                    p=2;
                    break;
                    case'(':
                    case')':
                    case'#':
                    p=1;
                    break;
                }
                return p;
            }
            int push(char item){
                 index++;
                 stack[index]=item;
            }
            int pop(){
                char a;
                a=stack[index];
                index--;
                return a;
            }
            void infToPos(char infix[],char postfix[]){
                int i,symbol,j=0;
                stack[++index]='#';
                for(i=0;infix[i]!='\0';i++){
                    symbol=infix[i];
                if(isoperator(symbol)==0)
                {
                    postfix[j]=symbol;
                    j++;
                }
                else{
                if(symbol=='(')
                    push(symbol);
                else if(symbol==')'){
                    while(stack[index]!='('){
                        postfix[j]=pop();
                        j++;
                    }
                    pop();
                } 
                else
                {
                    if(precedence(symbol)>precedence(stack[index]))
                    push(symbol);
                else
                 {
                    while(precedence(symbol)<=precedence(stack[index])){
                        postfix[j]=pop();
                        j++;
                    }
                    push(symbol);
                 }
                }
                }
                }
                while(stack[index]!='#'){
                    postfix[j]=pop();
                    j++;
                }
                postfix[j]='\0';
            
            }
            int main(){
                char infix[20],posfix[20];
                printf("Enter the infix expression");
                gets(infix);
                infToPos(infix,posfix);
                printf("postfix:");
                puts(posfix);
            }
