            #include <stdio.h>
            int push(char stack[][20],int n,int i){
                if (i==n-1)
                   return -1;
                else
                   i++;
                   printf("\nenter a value to push:");
                scanf("%s",&stack[i]);
                return i;
            }
            int pop(char stack[][20],int n,int i){
                if (i==-2)
                   return -1;
                else
                   i--;
                return i;
            }
            int peek(char stack[][20],int n,int i){
                if (i==-1)
                   return -1;
                else
                return i;
            }
