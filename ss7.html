Enter a Integer Array From the user, enter the integer value and perform binary search algorithm through recursion.

        
            #include<stdio.h>
            int binarysea(int[],int,int,int);
            void main(){
                int i,arr[10],u=10,l=0,key,position;
                printf("enter the elements in sort order:");
                for(i=0;i<10;i++){
                    scanf("%d",&arr[i]);   
                }
                printf("enter the element to be searched:");
                scanf("%d",&key);
                position=binarysea(arr,key,l,u);
                printf("Key is at %d position.",position);
                
            }
            int binarysea(int arr[],int key,int l,int u){
                int m;
                m=(l+u)/2;
                if(arr[m]==key)
                return (m+1);
                else if(arr[m]<key)
                binarysea(arr,key,m+1,u);
                else
                binarysea(arr,key,l,m+1);
            }
