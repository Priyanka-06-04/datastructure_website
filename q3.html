        //queue dynamically
            #include<stdio.h>

            struct node{
                int data;
                struct node *link;
            };
            struct node *front,*rear,*nn,*temp;
            int insert(int item){
                  nn=(struct node*)malloc(sizeof(struct node));
                  if(nn==NULL){
                    //printf("Overflow Condition");
                    return -1;
                  }
                else{
                    if(rear==NULL){
                    front=nn;
                    rear=nn; 
                    nn->data=item;
                    nn->link=NULL;  
                   }
                   else{
                    nn->data=item;
                    nn->link=NULL;
                    rear->link=nn;
                    rear=nn;
                   }
                   return 1;
                 } 
            }
                 int dequeue(){
                    if(front==NULL){

                   return -1;

                        front=front->link;
                        free (temp);
