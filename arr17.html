Merge the two sorted arrays into one array but in sorted array.

            
                #include<stdio.h>
                void mergesort(int arr1[],int size1,int arr2[],int size2,int arr3[]){
                    int i=0,k=0,j=0;
                    while(i<size1&&j<size2){
                    if(arr1[i]>arr2[j]){
                        arr3[k]=arr2[j];
                        j++;
                        k++;
                    }
                    else{
                        arr3[k]=arr1[i];
                        i++;
                        k++;
                    }
                    }
                    while(i<size1){
                        arr3[k]=arr1[i];
                        i++;
                        k++;
                    }
                    while(j<size2){
                        arr3[k]=arr2[j];
                        j++;
                        k++;
                    }
                }
