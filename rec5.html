                    int arr[20],i,size;

                    scanf("%d",&arr[i]);
                    printf("%d",max(arr,size-1));
                }
                int max(int arr[],int index){
                    int large;
                    if(index==0)
                    return arr[index];
                    large=max(arr,index-1);
                    if(arr[index]>large)
                    large=arr[index];
                    return large;
                }
