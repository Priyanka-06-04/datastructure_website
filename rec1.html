Calculate the factorial of number using Recursion

            
                //Factorial of Number
               #include<stdio.h>
            int main(){
                      int n;
                      printf("Enter the number");
                      scanf("%d",&n);
                      printf("The factorial of a number: %d",func(n));
                  }
                  int func(int n){
                     if(n==0)
                     return 1;
                     else
                     return n*func(n-1);
  
                  }

                Enter the number4
