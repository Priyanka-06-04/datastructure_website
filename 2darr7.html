Enter the matrix from user and copy it into another matrix

        #include <stdio.h>
            void matrix(int arr[3][2], int r, int c)
            {
                int i, j;
            
                for (i = 0; i < r; i++)
                {
                    printf("Enter the elements of %d row:", i + 1);
                    for (j = 0; j < c; j++)
                    {
                        scanf("%d", &(arr[i][j]));
                    }
                }
            }
            void copy(int arr[3][2],int arr2[3][2],int r,int c){
                int i,j;
                for(i=0;i<r;i++){
                    for(j=0;j<c;j++){
                         arr2[i][j]=arr[i][j];
                    }
                }
            }
            int main()
            {
                int arr[3][2],arr2[3][2],i,j;
                int rows = 3, cols = 2;
                matrix(arr, rows, cols);
                printf("The copied matrix is:\n");
                copy(arr,arr2,rows,cols);
                for (i = 0; i < rows; i++)
                {
                    for (j = 0; j < cols; j++)
                    {
                        printf("%d\t",arr2[i][j]);
                    }
                    printf("\n");
                }
                return 0;
            }
