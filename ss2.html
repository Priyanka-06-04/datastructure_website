Linear search through iteration

        
             #include<stdio.h>
             void main(){
                int arr[10],element,i;
                printf("enter the elements");
                for(i=0;i<10;i++){
                    scanf("%d",&arr[i]);   
                }
                
                printf("enter the element to be search");
                scanf("%d",&element);
                for( i=0;i<10;i++){
                    if(arr[i]==element){
                        printf("found at %d position",i+1);
                        break;
                    }
                    
                }
                if(i==10){
                    printf("not found");
                }
                
                }
