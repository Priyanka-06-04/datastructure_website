Enter a Integer Array From the user, enter the integer value and perform linear search algorithm through recursion.

            
            #include<stdio.h>
                int search(int[],int,int);
                void main(){
                    int arr[10],element,i,size;
                    printf("enter the size");
                    scanf("%d",&size);
                    printf("enter the elements");
                    for(i=0;i<size;i++){
                        scanf("%d",&arr[i]);   
                    }
                    printf("enter the element to be search");
                    scanf("%d",&element);
                    i=search(arr,10,element);
                    if(i==-1){
                        printf("not found");
                    }
                    else
                    printf("found at %d",i+1);
                }
                    int search(int arr[],int size,int element){
                        if(size<-1)
                        return -1;
                        else if(arr[size]==element)
                        return size;
                        else
                        return search(arr,size-1,element);
                }
