                //convert into uppercase
                #include<stdio.h>
                void main(){
                    char str[20];
                    int i=0;
                    printf("Enter the name in lowercase: ");
                    gets(str);
                    for(i=0;str[i]!='\0';i++){
                         if(str[i]>='a' && str[i]<='z'){
                           str[i]=str[i]-32;
                           }     
                    } 
                    printf("After convert into uppercase\n");
                    printf("%s",str);  
                }
