            void mergesort(int arr[],int l,int u){
                int mid;
                if(l<u){
                mid=((l+u)/2);
            
                mergesort(arr,l,mid);
                mergesort(arr,mid+1,u);
                merge(arr,l,mid,u);
                }
            }
            void merge(int arr[],int l,int m,int u){
