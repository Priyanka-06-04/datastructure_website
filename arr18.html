                    int size,arr[10],l,u,i;
                    printf("Enter the size of array: ");

                    printf("Enter the elements ");
                    for(i=0;i<size;i++){
                        scanf("\t%d",&arr[i]);
                    }
                    printf("Local minima numbers in array: \n");
                    for(i=1;i<size-1;i++){
                     l=arr[i-1];
                     u=arr[i+1];
                     if(arr[i]<l && arr[i]<u)
                     printf("%d\t",arr[i]);
                    }
                }
