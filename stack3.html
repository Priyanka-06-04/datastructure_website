Create a stack implementing the operations on an integer array(dynamically)(push,pop,peek,isempty,isfull)

        #include<stdio.h>
            #include<conio.h>
            #include<stdlib.h>
            struct node{
                int data;
                struct node *next;
            };
            struct node *start,*nn,*temp;
            int push(int value){
                if(start==NULL){
                start=(struct node*)malloc(sizeof(struct node));
                start->data=value;
                start->next=NULL;
                 }
                 else{
                    nn=(struct node*)malloc(sizeof(struct node));
                    nn->data=value;
                    nn->next=start;
                    start=nn;
                 }
            }
            int pop(int value){
                if(start==NULL)
                printf("underflow");
                else{
                temp=start;
                value=start->data;
                start=start->next;
                free(temp);
                }
                return value;
            }
            int peek(){
                printf("%d",start->data);
            }
            int isEmpty(){
                if(start==NULL)
                printf("underflow");
                else
                printf("stack is not empty");
            }
            int isFull(){
                nn=(struct node*)malloc(sizeof(struct node));
                if(nn==NULL)
                printf("Overflow");
                else
                printf("Stack is not Full");
            }
            int show(){
                temp=start;
                 printf("stack is.......");
                while(temp!=NULL){
                    printf("%d",temp->data);
                    temp=temp->next;
                }
            }
