Calculate the fibonacci of number using Recursion

            //Fibonacci series
                #include<stdio.h>
                int fib(int);
                int main(){
                    int n,i;
                    printf("Enter the number");
                    scanf("%d",&n);
                    printf("Fibonacci series....");
                    for(i=0;i<n;i++)
                    printf("%d ",fib(i));
                }
                int fib(int n){
                    if(n==0)
                    return 0;
                    if(n==1)
                    return 1;
                    else
                    return (fib(n-1)+fib(n-2));
                }
