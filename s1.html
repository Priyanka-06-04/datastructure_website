Enter a Integer Array From the user,perform selection sort algorithm through iteration.

        //selection sort 
            #include<stdio.h>
            int main()
            {
            int arr[5];
            printf("Enter 5 elements of the array:");
            for(int i=0;i<5;i++)
            scanf("%d",&arr[i]);
            int min,swap;
            for(int i=0;i<4;i++){
                min=i;
                for(int j=i+1;j<5;j++){
                    if(arr[min]>arr[j])
                    min=j;
                }
                swap=arr[min];
                arr[min]=arr[i];
                arr[i]=swap;
            }
            for(int i=0;i<5;i++){
                printf("%d\t",arr[i]);
            }
            return 0;
            }
