Convert into Lowercase Case

            
                //convert into lowerercase
                #include<stdio.h>
                void main(){
                    char str[20];
                    int i=0;
                    printf("Enter the name in Uppercase: ");
                    gets(str);
                    for(i=0;str[i]!='\0';i++){
                         if(str[i]>='A' && str[i]<='Z'){
                           str[i]=str[i]+32;
                           }     
                    } 
                    printf("After convert into lowercase\n");
                    printf("%s",str);  
                }
